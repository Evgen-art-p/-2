import React, { useState, useEffect } from 'react';
import { 
  Radio, 
  History, 
  Users, 
  Settings, 
  Send, 
  Bell, 
  Heart, 
  Zap, 
  ShieldCheck,
  ChevronRight,
  Info
} from 'lucide-react';

const App = () => {
  const [activeTab, setActiveTab] = useState('live');
  const [tutorMessage, setTutorMessage] = useState('');
  const [feed, setFeed] = useState([
    { id: 1, time: '22:05', type: 'resonance', text: 'Резонанс: Артем смеется. Эйрик рассказал историю про «щекочущий мох».', icon: <Heart className="w-4 h-4 text-pink-500" /> },
    { id: 2, time: '22:12', type: 'artifact', text: 'Хронос: Зафиксирован новый артефакт «Перо горного орла».', icon: <Zap className="w-4 h-4 text-yellow-400" /> },
    { id: 3, time: '22:18', type: 'sofia', text: 'София: Эмоциональный фон стабилен. Глубина погружения — 8/10.', icon: <ShieldCheck className="w-4 h-4 text-cyan-400" /> },
  ]);
  const [notification, setNotification] = useState(null);

  // Симуляция живого потока
  useEffect(() => {
    const timer = setTimeout(() => {
      const newMessage = {
        id: Date.now(),
        time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
        type: 'resonance',
        text: 'Эйрик: "Смотри, Артем, это северное сияние Грондхейма!"',
        icon: <Info className="w-4 h-4 text-purple-400" />
      };
      setFeed(prev => [newMessage, ...prev].slice(0, 5));
    }, 15000);
    return () => clearTimeout(timer);
  }, [feed]);

  const showNotification = (msg) => {
    setNotification(msg);
    setTimeout(() => setNotification(null), 3000);
  };

  const handleSendToSpark = () => {
    if (!tutorMessage.trim()) return;
    showNotification("Данные переданы Нейро Спарку. Сюжет адаптируется...");
    setTutorMessage('');
  };

  const renderContent = () => {
    switch (activeTab) {
      case 'live':
        return (
          <div className="space-y-6 animate-in fade-in duration-500">
            {/* Статус-орб */}
            <div className="flex flex-col items-center justify-center py-8">
              <div className="relative">
                <div className="w-24 h-24 rounded-full bg-cyan-500/20 flex items-center justify-center animate-pulse border border-cyan-500/30 shadow-[0_0_50px_rgba(6,182,212,0.3)]">
                  <div className="w-16 h-16 rounded-full bg-cyan-400/40 flex items-center justify-center">
                    <div className="w-8 h-8 rounded-full bg-cyan-400 shadow-[0_0_20px_rgba(34,211,238,1)]"></div>
                  </div>
                </div>
                <div className="absolute -top-2 -right-2 bg-red-500 text-[10px] px-2 py-0.5 rounded-full animate-bounce">LIVE</div>
              </div>
              <div className="mt-6 text-center">
                <p className="text-cyan-400 font-bold tracking-widest text-sm uppercase">Связь: Артем + Эйрик</p>
                <p className="text-gray-500 text-[10px] mt-1 uppercase tracking-tighter">3D-Audio Stream Active • Ultra-Low Latency</p>
              </div>
            </div>

            {/* Лента событий */}
            <div>
              <div className="flex justify-between items-center mb-4">
                <h3 className="text-gray-400 text-sm font-medium flex items-center gap-2">
                  <Bell className="w-4 h-4" /> События Души
                </h3>
              </div>
              <div className="space-y-3">
                {feed.map((item) => (
                  <div key={item.id} className="bg-[#1a1a20] border-l-2 border-purple-500 p-4 rounded-r-lg shadow-sm group hover:bg-[#202028] transition-colors">
                    <div className="flex justify-between items-start mb-1">
                      <span className="text-[10px] text-gray-600 font-mono">{item.time}</span>
                      {item.icon}
                    </div>
                    <p className="text-sm text-gray-300 leading-relaxed">{item.text}</p>
                  </div>
                ))}
              </div>
            </div>

            {/* Тьютор Линк */}
            <div className="bg-gradient-to-b from-[#1a1a20] to-[#16161a] p-5 rounded-2xl border border-dashed border-cyan-500/40 relative overflow-hidden">
               <div className="absolute top-0 right-0 p-2 opacity-10">
                  <Send className="w-12 h-12 text-cyan-500" />
               </div>
               <label className="block text-cyan-400 text-xs font-bold uppercase mb-3 tracking-widest">Тьютор Линк (Контекст дня)</label>
               <textarea 
                value={tutorMessage}
                onChange={(e) => setTutorMessage(e.target.value)}
                className="w-full bg-black/50 border border-gray-800 rounded-xl p-3 text-sm text-white placeholder:text-gray-700 focus:ring-1 focus:ring-cyan-500 outline-none transition-all"
                rows="3"
                placeholder="Расскажите Спарку о важном событии или планах на завтра..."
               ></textarea>
               <button 
                onClick={handleSendToSpark}
                className="w-full mt-4 bg-transparent border border-cyan-500 text-cyan-400 py-3 rounded-xl text-xs font-bold uppercase tracking-widest hover:bg-cyan-500 hover:text-black transition-all active:scale-95 shadow-[0_0_15px_rgba(6,182,212,0.1)]"
               >
                 Внедрить в мир книги
               </button>
            </div>
          </div>
        );
      case 'memory':
        return (
          <div className="space-y-4 py-6 animate-in slide-in-from-right duration-300">
            <h2 className="text-xl font-light text-white">Хроники Грондхейма</h2>
            <p className="text-gray-500 text-sm">Здесь хранятся записи всех путешествий.</p>
            {[1, 2, 3].map(i => (
              <div key={i} className="bg-[#16161a] p-4 rounded-xl border border-gray-800 flex justify-between items-center">
                <div>
                  <div className="text-xs text-purple-400 mb-1">Глава {i}: Пробуждение</div>
                  <div className="text-white text-sm">Сессия от {10 + i} марта</div>
                </div>
                <ChevronRight className="text-gray-600 w-5 h-5" />
              </div>
            ))}
          </div>
        );
      case 'heroes':
        return (
          <div className="space-y-6 py-6 animate-in slide-in-from-right duration-300">
            <div className="bg-gradient-to-r from-purple-900/20 to-cyan-900/20 p-6 rounded-2xl border border-white/5 flex items-center gap-6">
               <div className="w-20 h-20 rounded-full bg-gray-800 border-2 border-cyan-500 flex items-center justify-center overflow-hidden">
                  <div className="text-3xl">🛡️</div>
               </div>
               <div>
                  <h2 className="text-2xl font-bold text-white uppercase tracking-tighter">Эйрик</h2>
                  <p className="text-cyan-400 text-xs uppercase font-mono">Верный Хранитель • Уровень 4</p>
               </div>
            </div>
            <div className="grid grid-cols-2 gap-4">
              <div className="bg-[#16161a] p-4 rounded-xl border border-gray-800 text-center">
                <p className="text-gray-500 text-[10px] uppercase">Связь</p>
                <p className="text-xl font-bold text-white">88%</p>
              </div>
              <div className="bg-[#16161a] p-4 rounded-xl border border-gray-800 text-center">
                <p className="text-gray-500 text-[10px] uppercase">Знания</p>
                <p className="text-xl font-bold text-white">12/40</p>
              </div>
            </div>
          </div>
        );
      default:
        return null;
    }
  };

  return (
    <div className="min-h-screen bg-[#0a0a0c] text-gray-200 font-sans selection:bg-cyan-500/30">
      {/* Header */}
      <header className="sticky top-0 z-50 bg-[#0a0a0c]/80 backdrop-blur-md border-b border-gray-800 px-6 py-4 flex flex-col items-center shadow-lg">
        <div className="text-[10px] tracking-[0.3em] text-purple-500 font-black uppercase mb-1">Studio Six Fingers</div>
        <h1 className="text-lg font-light tracking-tight text-white italic">Living Book: <span className="text-cyan-400 font-normal">Грондхейм</span></h1>
      </header>

      {/* Main Container */}
      <main className="max-w-md mx-auto px-6 pb-32">
        {renderContent()}
      </main>

      {/* Toast Notification */}
      {notification && (
        <div className="fixed top-20 left-1/2 -translate-x-1/2 z-50 bg-cyan-500 text-black px-6 py-2 rounded-full text-xs font-bold shadow-[0_0_20px_rgba(6,182,212,0.4)] animate-in fade-in zoom-in duration-200 uppercase tracking-widest">
          {notification}
        </div>
      )}

      {/* Navigation Footer */}
      <nav className="fixed bottom-0 left-0 right-0 bg-[#0a0a0c] border-t border-gray-800 px-6 py-4 flex justify-around items-center z-50">
        <NavItem 
          active={activeTab === 'live'} 
          onClick={() => setActiveTab('live')} 
          icon={<Radio size={20} />} 
          label="ЭФИР" 
        />
        <NavItem 
          active={activeTab === 'memory'} 
          onClick={() => setActiveTab('memory')} 
          icon={<History size={20} />} 
          label="ПАМЯТЬ" 
        />
        <NavItem 
          active={activeTab === 'heroes'} 
          onClick={() => setActiveTab('heroes')} 
          icon={<Users size={20} />} 
          label="ГЕРОИ" 
        />
        <NavItem 
          active={activeTab === 'settings'} 
          onClick={() => setActiveTab('settings')} 
          icon={<Settings size={20} />} 
          label="ОПЦИИ" 
        />
      </nav>

      {/* Loca Tip (Floating) */}
      <div className="fixed bottom-24 right-6 w-10 h-10 bg-purple-600 rounded-full flex items-center justify-center shadow-lg border border-purple-400 group cursor-pointer overflow-hidden">
          <div className="absolute inset-0 bg-white/20 animate-pulse"></div>
          <span className="text-white text-[8px] font-bold z-10 group-hover:hidden">LOCA</span>
          <Info className="hidden group-hover:block text-white w-5 h-5 z-10" />
          <div className="absolute bottom-full right-0 mb-4 w-48 bg-[#1a1a20] border border-purple-500 p-3 rounded-xl opacity-0 group-hover:opacity-100 transition-opacity pointer-events-none text-[10px] text-gray-300">
             "Помни: Спарк лучше всего реагирует на детали. Расскажи ему, что сегодня на ужин!"
          </div>
      </div>
    </div>
  );
};

const NavItem = ({ active, onClick, icon, label }) => (
  <button 
    onClick={onClick}
    className={`flex flex-col items-center gap-1 transition-all ${active ? 'text-cyan-400 scale-110' : 'text-gray-600 hover:text-gray-400'}`}
  >
    {icon}
    <span className="text-[9px] font-bold tracking-tighter">{label}</span>
    {active && <div className="w-1 h-1 rounded-full bg-cyan-400 mt-0.5 shadow-[0_0_8px_rgba(34,211,238,1)]"></div>}
  </button>
);

export default App;
